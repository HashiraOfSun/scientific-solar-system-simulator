cmake_minimum_required(VERSION 3.20)
project(ScientificSolarSystem VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(glm CONFIG REQUIRED)
add_library(OrbitalModel Planet.cpp Animation.cpp)
target_include_directories(OrbitalModel PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(OrbitalModel PUBLIC glm::glm)
target_compile_options(OrbitalModel PRIVATE -Wall -Wextra -Wpedantic)
option(BUILD_VIEWER "Build macOS OpenGL 4.1 viewer" ON)
if(BUILD_VIEWER)
    if(NOT APPLE)
        message(FATAL_ERROR "Viewer currently targets macOS. Use -DBUILD_VIEWER=OFF for portable model/tests.")
    endif()
    find_package(glfw3 3.3 CONFIG REQUIRED)
    find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
    add_executable(SolarSystem main.cpp Shader.cpp Texture.cpp Hud.cpp)
    target_compile_definitions(SolarSystem PRIVATE GL_SILENCE_DEPRECATION GLFW_INCLUDE_NONE)
    target_compile_options(SolarSystem PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(SolarSystem PRIVATE OrbitalModel glfw ${OPENGL_FRAMEWORK})
    # Refresh assets on every configure, including edits to a shader without C++ changes.
    file(COPY shaders textures DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
    add_custom_command(TARGET SolarSystem POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:SolarSystem>/shaders
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/textures $<TARGET_FILE_DIR:SolarSystem>/textures)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(ScienceTests tests/science_tests.cpp)
    target_link_libraries(ScienceTests PRIVATE OrbitalModel)
    target_compile_options(ScienceTests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME science COMMAND ScienceTests)
endif()
